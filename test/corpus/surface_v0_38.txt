@vectorize
fn total(xs: bounded[Int; 1024]) -> Int {
    let mut s = 0;
    for x in xs {
        s = s + x;
    }
    return s;
}

fn bytes(xs: bounded[U8; 1024]) -> U8 {
    return xs.map(it).sum(U8(0));

    params { k: Int = 0; }
    @hot @vectorize
    fn chained(xs: bounded[Int; 1024]) -> Int {
        return xs.map(it + 1).sum(0);
